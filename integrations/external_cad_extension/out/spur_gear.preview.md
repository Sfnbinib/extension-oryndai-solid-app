# Macro Preview: spur_gear

Prompt: Create a 24 tooth spur gear, module 2, thickness 8 mm, bore 8 mm.

## Operation Steps

1. `sketch` `gear_blank_sketch` - Sketch simplified gear blank
2. `circle` `gear_outer` - Outer gear diameter
3. `extrude` `gear_body` - Extrude gear thickness
4. `hole` `gear_bore` - Central bore
5. `pattern` `tooth_pattern` - Declare tooth pattern for future involute generator
6. `chamfer` `tooth_edge_break` - Small edge break
7. `export` `export_step` - Export editable STEP

## Assumptions

- Pitch diameter is module * teeth = 48 mm.
- Outer diameter is pitch diameter + 2 * module = 52 mm.
- The first prototype declares a tooth pattern but does not generate true involute geometry in VBA.

## Warnings

- Use a true involute tooth adapter before manufacturing or mating with real gears.

## Validation

- ok: `true`

Dim swApp As Object
Dim swModel As Object

Sub main()
    Set swApp = Application.SldWorks
    Set swModel = swApp.NewPart
    If swModel Is Nothing Then
        MsgBox "Could not create SolidWorks part document."
        Exit Sub
    End If
    swModel.Extension.CustomPropertyManager("").Add3 "ORYNDPlan", 30, "spur_gear", 2

    ' gear_blank_sketch: Sketch simplified gear blank
    ORYND_CreateSketch swModel, "Top"
    ' gear_outer: Outer gear diameter
    ORYND_Circle swModel, 0.000000000, 0.000000000, 0.026000000
    ' gear_body: Extrude gear thickness
    ORYND_Extrude swModel, 0.008000000, False
    ' gear_bore: Central bore
    ORYND_Hole swModel, 0.000000000, 0.000000000, 0.008000000, True, 0.000000000
    ' tooth_pattern: Declare tooth pattern for future involute generator
    ORYND_Pattern swModel, "circular", 24, 0.026000000, "Z", "gear_outer"
    ' tooth_edge_break: Small edge break
    ORYND_Chamfer swModel, 0.000400000, "tooth_edges"
    ' export_step: Export editable STEP
    ORYND_Export swModel, "STEP", "spur_gear.step"
End Sub

Private Sub ORYND_CreateSketch(ByVal model As Object, ByVal planeName As String)
    Dim ok As Boolean
    ok = False
    If planeName = "Front" Then ok = model.Extension.SelectByID2("Front Plane", "PLANE", 0, 0, 0, False, 0, Nothing, 0)
    If planeName = "Top" Then ok = model.Extension.SelectByID2("Top Plane", "PLANE", 0, 0, 0, False, 0, Nothing, 0)
    If planeName = "Right" Then ok = model.Extension.SelectByID2("Right Plane", "PLANE", 0, 0, 0, False, 0, Nothing, 0)
    If ok Then model.SketchManager.InsertSketch True
End Sub

Private Sub ORYND_Circle(ByVal model As Object, ByVal x As Double, ByVal y As Double, ByVal radius As Double)
    model.SketchManager.CreateCircleByRadius x, y, 0#, radius
End Sub

Private Sub ORYND_Rectangle(ByVal model As Object, ByVal x As Double, ByVal y As Double, ByVal width As Double, ByVal height As Double)

End Sub
